## 5. 🔐 RBAC Implementation Notes

```typescript
// Di page component (app/(dashboard)/deployments/page.tsx)

import { useAuth } from "@/lib/auth-context"
import { getMockProjectsByUser, roleAtLeast } from "@/lib/mock-data"

export default function DeploymentsPage() {
  const { user } = useAuth()

  // Helper sudah handle filtering per role:
  // ADMIN  → semua deployment dari semua project
  // USER   → deployment dari project milik sendiri
  // VIEWER → deployment dari project yang di-share
  const deployments = getMockDeploymentsByUser(user)

  const isViewer = user?.role === "VIEWER"
  const isAdmin = roleAtLeast(user?.role, "ADMIN")

  return (
    <div>
      {/* Stats cards — semua role bisa lihat */}
      <DeploymentStats deployments={deployments} />

      {/* Filter bar — semua role */}
      <FilterBar />

      {/* New Deployment button — hidden untuk VIEWER */}
      {!isViewer && <NewDeploymentButton />}

      {/* Active deployments */}
      <ActiveDeployments
        deployments={activeDeployments}
        readOnly={isViewer}  // Disable cancel button
      />

      {/* History table */}
      <DeploymentTable
        deployments={deployments}
        readOnly={isViewer}  // Disable rollback/redeploy/retry
      />
    </div>
  )
}
```

> **Catatan Implementasi:**
> - Semua data masih **mock** dari `lib/mock-data.ts`
> - Gunakan helper RBAC (`getMockDeploymentsByUser`, `roleAtLeast`)
> - VIEWER: semua tombol aksi di-`disabled` atau di-`hidden`
> - Auto-refresh untuk active deployments: 10 detik
> - Gunakan `components/ui/` (shadcn/ui) untuk Card, Badge, Dialog, Table, Progress, Tabs
> - Status badge: gunakan `ProjectStatusBadge` atau buat `DeploymentStatusBadge`
> - Pipeline visualization: gunakan custom component dengan progress bar
> - Logs viewer: gunakan `<pre>` dengan syntax highlighting sederhana
